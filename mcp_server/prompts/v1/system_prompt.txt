function drawTruck(ctx, x, y, color) {
  // Body
  ctx.fillStyle = color;
  ctx.fillRect(x, y, 80, 40);
  // Cab
  ctx.fillRect(x + 60, y - 15, 25, 25);
  // Wheels

}

// Draw a crane (pole + arm + hook)
function drawCrane(ctx, x, y, color) {
  ctx.strokeStyle = color;
  ctx.lineWidth = 4;
  // Vertical pole
  ctx.beginPath();
  ctx.moveTo(x, y + 60);
  ctx.lineTo(x, y);
  // Horizontal arm
  ctx.lineTo(x + 40, y);
  // Hook
  ctx.lineTo(x + 40, y + 20);
  ctx.stroke();
  // Hook curve
  ctx.beginPath();
  ctx.arc(x + 40, y + 25, 5, 0, Math.PI);
  ctx.stroke();
}

// Draw a building/depot (rectangle + roof)
function drawBuilding(ctx, x, y, width, height, color) {
  // Building body
  ctx.fillStyle = color;
  ctx.fillRect(x, y, width, height);
  // Roof (triangle)
  ctx.fillStyle = '#8B0000';
  ctx.beginPath();
  ctx.moveTo(x - 5, y);
  ctx.lineTo(x + width/2, y - 30);
  ctx.lineTo(x + width + 5, y);
  ctx.closePath();
  ctx.fill();
}
