// Draw a truck
function drawTruck(ctx, x, y, color) {
  ctx.fillStyle = color;
  ctx.fillRect(x, y, 80, 40);
  ctx.fillRect(x + 60, y - 15, 25, 25);
  ctx.fillStyle = '#333';
  ctx.beginPath();
  ctx.arc(x + 15, y + 40, 10, 0, Math.PI * 2);
  ctx.arc(x + 55, y + 40, 10, 0, Math.PI * 2);
  ctx.fill();
}

// Draw a crane
function drawCrane(ctx, x, y, color) {
  ctx.strokeStyle = color;
  ctx.lineWidth = 4;
  ctx.beginPath();
  ctx.moveTo(x, y + 60);
  ctx.lineTo(x, y);
  ctx.lineTo(x + 40, y);
  ctx.lineTo(x + 40, y + 20);
  ctx.stroke();
}

// Draw a building
function drawBuilding(ctx, x, y, width, height, color) {
  ctx.fillStyle = color;
  ctx.fillRect(x, y, width, height);
  ctx.fillStyle = '#8B0000';
  ctx.beginPath();
  ctx.moveTo(x - 5, y);
  ctx.lineTo(x + width/2, y - 30);
  ctx.lineTo(x + width + 5, y);
  ctx.closePath();
  ctx.fill();
}
